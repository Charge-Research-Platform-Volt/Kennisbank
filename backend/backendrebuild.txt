  using Azure.AI.DocumentIntelligence;
  using Azure;
  using UglyToad.PdfPig;
  using System.Text;

  namespace KnowledgeBank.Services;

  /// <summary>
  /// Service for extracting text from various document formats
  /// Uses free methods (PdfPig) when possible, falls back to Azure Document Intelligence
  /// </summary>
  public class TextExtractionService
  {
      private readonly ILogger<TextExtractionService> _logger;
      private readonly DocumentIntelligenceClient _docIntelligenceClient;

      public TextExtractionService(
          ILogger<TextExtractionService> logger,
          DocumentIntelligenceClient docIntelligenceClient)
      {
          _logger = logger;
          _docIntelligenceClient = docIntelligenceClient;
      }

      /// <summary>
      /// Extracts text from a document stream, choosing the best extraction method based on file type
      /// </summary>
      /// <param name="stream">Document file stream</param>
      /// <param name="fileExtension">File extension (e.g., ".pdf", ".docx")</param>
      /// <returns>Extracted text content</returns>
      public async Task<string> ExtractTextAsync(Stream stream, string fileExtension)
      {
          fileExtension = fileExtension.ToLowerInvariant();

          _logger.Information("Extracting text from {FileType} document", fileExtension);

          try
          {
              if (fileExtension == ".pdf")
              {
                  // Try free PdfPig extraction first
                  return await ExtractFromPdfWithFallback(stream);
              }
              else if (fileExtension == ".txt")
              {
                  // Plain text - direct read
                  return await ExtractFromPlainText(stream);
              }
              else
              {
                  // For DOCX, XLSX, PPTX, etc. - use Azure DI Read
                  return await ExtractWithAzureDocumentIntelligence(stream, fileExtension);
              }
          }
          catch (Exception ex)
          {
              _logger.Error(ex, "Failed to extract text from {FileType}", fileExtension);
              throw;
          }
      }

      /// <summary>
      /// Extracts text from PDF using free PdfPig, falls back to Azure DI Read if needed
      /// </summary>
      private async Task<string> ExtractFromPdfWithFallback(Stream stream)
      {
          long originalPosition = stream.Position;

          try
          {
              // Attempt free extraction with PdfPig
              _logger.Information("Attempting free PDF text extraction with PdfPig");

              stream.Position = 0;
              var text = ExtractWithPdfPig(stream);

              // Validate extraction quality
              if (!string.IsNullOrWhiteSpace(text) && text.Length > 50)
              {
                  _logger.Information("PdfPig extraction successful: {Length} characters", text.Length);
                  return text;
              }

              // If extraction resulted in very little text, might be scanned/image PDF
              _logger.Warning("PdfPig extracted minimal text ({Length} chars), falling back to Azure DI Read (OCR)",
                  text?.Length ?? 0);

              stream.Position = 0;
              return await ExtractWithAzureDocumentIntelligence(stream, ".pdf");
          }
          catch (Exception ex)
          {
              // PdfPig failed (corrupted PDF, encrypted, etc.) - fallback to Azure DI
              _logger.Warning(ex, "PdfPig extraction failed, falling back to Azure Document Intelligence Read");

              stream.Position = 0;
              return await ExtractWithAzureDocumentIntelligence(stream, ".pdf");
          }
          finally
          {
              stream.Position = originalPosition;
          }
      }

      /// <summary>
      /// Extracts text from PDF using free PdfPig library
      /// </summary>
      private string ExtractWithPdfPig(Stream stream)
      {
          var textBuilder = new StringBuilder();

          using (var document = PdfDocument.Open(stream))
          {
              foreach (var page in document.GetPages())
              {
                  var pageText = page.Text;
                  if (!string.IsNullOrWhiteSpace(pageText))
                  {
                      textBuilder.AppendLine(pageText);
                      textBuilder.AppendLine(); // Page separator
                  }
              }
          }

          return textBuilder.ToString();
      }

      /// <summary>
      /// Extracts text using Azure Document Intelligence "prebuilt-read" model
      /// Supports PDF, DOCX, XLSX, PPTX, images (with OCR), and more
      /// </summary>
      private async Task<string> ExtractWithAzureDocumentIntelligence(Stream stream, string fileExtension)
      {
          _logger.Information("Using Azure Document Intelligence Read model for {FileType}", fileExtension);

          var operation = await _docIntelligenceClient.AnalyzeDocumentAsync(
              WaitUntil.Completed,
              "prebuilt-read",  // Cheaper "read" model instead of "layout"
              stream,
              outputContentFormat: Azure.AI.DocumentIntelligence.ContentFormat.Text
          );

          var extractedText = operation.Value.Content ?? string.Empty;

          _logger.Information("Azure DI extracted {Length} characters from {FileType}",
              extractedText.Length, fileExtension);

          return extractedText;
      }

      /// <summary>
      /// Extracts text from plain text file
      /// </summary>
      private async Task<string> ExtractFromPlainText(Stream stream)
      {
          stream.Position = 0;
          using var reader = new StreamReader(stream, leaveOpen: true);
          return await reader.ReadToEndAsync();
      }
  }

  [HttpPost("extract-metadata")]
  [SwaggerOperation(Summary = "Extract metadata and tags from document without saving")]
  [SwaggerResponse(200, "Metadata extracted successfully", typeof(MetadataExtractionResponseDto))]
  [SwaggerResponse(400, "Bad Request", typeof(MetadataExtractionResponseDto))]
  [SwaggerResponse(500, "Internal Server Error", typeof(MetadataExtractionResponseDto))]
  public async Task<IActionResult> ExtractMetadata([FromForm] IFormFile file)
  {
      try
      {
          logger.Information("Metadata extraction request received for file: {FileName}", file?.FileName);

          // Validate file
          if (file == null || file.Length == 0)
          {
              return BadRequest(new MetadataExtractionResponseDto
              {
                  Success = false,
                  Message = "No file uploaded"
              });
          }

          var fileExtension = Path.GetExtension(file.FileName).ToLowerInvariant();

          // Supported file types
          var allowedExtensions = new[] { ".pdf", ".docx", ".doc", ".txt", ".rtf", ".pptx", ".xlsx" };
          if (!allowedExtensions.Contains(fileExtension))
          {
              return BadRequest(new MetadataExtractionResponseDto
              {
                  Success = false,
                  Message = $"Unsupported file type '{fileExtension}'. Supported: {string.Join(", ", allowedExtensions)}"
              });
          }

          // Check file size
          const long maxFileSize = 100 * 1024 * 1024; // 100 MB
          if (file.Length > maxFileSize)
          {
              return BadRequest(new MetadataExtractionResponseDto
              {
                  Success = false,
                  Message = $"File too large. Maximum size: {maxFileSize / (1024 * 1024)} MB"
              });
          }

          logger.Information("Extracting text from {FileType} document", fileExtension);

          // Extract text using unified service (PdfPig for PDF, Azure DI for others)
          string extractedText;
          using (var stream = file.OpenReadStream())
          {
              extractedText = await _textExtractionService.ExtractTextAsync(stream, fileExtension);
          }

          // Validate extracted text
          if (string.IsNullOrWhiteSpace(extractedText))
          {
              return BadRequest(new MetadataExtractionResponseDto
              {
                  Success = false,
                  Message = "No text could be extracted from the document. The file may be empty, corrupted, or image-only."
              });
          }

          logger.Information("Text extracted successfully. Length: {TextLength} characters", extractedText.Length);

          // Extract metadata and tags using LLM
          logger.Information("Analyzing document with LLM to extract metadata and tags");

          var metadata = await ragManager.ExtractMetadataAndTagsAsync(extractedText, file.FileName);

          if (metadata == null)
          {
              return StatusCode(500, new MetadataExtractionResponseDto
              {
                  Success = false,
                  Message = "Failed to extract metadata and tags from document."
              });
          }

          logger.Information("Metadata extraction completed successfully for {FileName}", file.FileName);

          return Ok(new MetadataExtractionResponseDto
          {
              Success = true,
              Message = "Metadata and tags extracted successfully",
              Data = new MetadataExtractionData
              {
                  Title = metadata.Title,
                  Abstract = metadata.Abstract,
                  Description = metadata.Description,
                  PublicationDate = metadata.PublicationDate,
                  LanguageCode = metadata.LanguageCode,
                  Authors = metadata.Authors,
                  PublicationCode = metadata.PublicationCode,
                  Keywords = metadata.Keywords,
                  Tags = metadata.Tags
              }
          });
      }
      catch (Exception ex)
      {
          logger.Error(ex, "Error extracting metadata from file: {FileName}", file?.FileName);
          return StatusCode(500, new MetadataExtractionResponseDto
          {
              Success = false,
              Message = $"Internal error during metadata extraction: {ex.Message}"
          });
      }
  }
